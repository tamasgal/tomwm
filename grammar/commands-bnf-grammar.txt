# EBNF-like grammar of tomwm CLI args to generate shell completion. Managed by https://github.com/adaszko/complgen
# Run ./build-shell-completion.sh to generate shell completion
tomwm <subcommand>;
tomwm -v;
tomwm --version;
tomwm --help;
tomwm -h;

<subcommand> ::=

    close [--quit-if-last-window|--window-id <window_id>]...

    | close-all-windows-but-current [--quit-if-last-window]

    | config [--json | --keys]... --get {{{ tomwm config --major-keys }}} [--json | --keys]...
    | config --major-keys
    | config --all-keys
    | config --config-path

    | debug-windows [--window-id <window_id>]

    | echo [--stderr|--window <window_id>]... --

    | enable toggle
    | enable [--fail-if-noop] on [--fail-if-noop]
    | enable [--fail-if-noop] off [--fail-if-noop]

    | eval [--stdin]

    | false

    | floating [--fail-if-noop|--window-id <window_id>]... [on|off] [--fail-if-noop|--window-id <window_id>]...

    | focus [<focus_direction_flag>]... (left|down|up|right) [<focus_direction_flag>]...
    | focus [--ignore-floating|--fail-if-noop]... (next|prev) [--ignore-floating|--fail-if-noop]...
    | focus --window-id <window_id>

    | focus-back-and-forth

    | focus-monitor [--wrap-around] (left|down|up|right) [--wrap-around]
    | focus-monitor [--wrap-around] (next|prev) [--wrap-around]
    | focus-monitor <monitor_pattern>...

    | fullscreen [--no-outer-gaps|--window-id <window_id>]...
    | fullscreen [--no-outer-gaps|--fail-if-noop|--window-id <window_id>]... on [--no-outer-gaps|--fail-if-noop|--window-id <window_id>]...
    | fullscreen [--fail-if-noop|--window-id <window_id>]... off [--fail-if-noop|--window-id <window_id>]...

    | layout [--monitor <monitor_id>|--fail-if-noop]... (<layouts>)... [--monitor <monitor_id>|--fail-if-noop]...
    | layout [--monitor <monitor_id>] --toggle [--monitor <monitor_id>]

    | macos-native-fullscreen [--fail-if-noop|--window-id <window_id>]... [on|off] [--fail-if-noop|--window-id <window_id>]...

    | macos-native-minimize [--window-id <window_id>]

    | mfact [--monitor <monitor_id>] <number>

    | mode <mode_id>

    | move [--window-id <window_id>|--fail-if-noop]... (next|prev) [--window-id <window_id>|--fail-if-noop]...

    | move-mouse [--fail-if-noop] (monitor-lazy-center|monitor-force-center|window-lazy-center|window-force-center) [--fail-if-noop]

    | move-to-monitor [<move_to_monitor1_flag>]... (left|down|up|right|next|prev) [<move_to_monitor1_flag>]...
    | move-to-monitor [<move_to_monitor2_flag>]... <monitor_pattern>... [<move_to_monitor2_flag>]...

    | nmaster [--monitor <monitor_id>] <number>

    | reload-config [--no-gui | --dry-run]...

    | run-callback [--for-every-window|--window-id <window_id>]... on-window-detected [--for-every-window|--window-id <window_id>]...
    | run-callback (on-focus-changed|on-focused-monitor-changed)

    | tag [--toggle|--window-id <window_id>]... <tag>... [--toggle|--window-id <window_id>]...
    | tag [--toggle|--window-id <window_id>]... --index <number>... [--toggle|--window-id <window_id>]...
    | tag [--window-id <window_id>] --all [--window-id <window_id>]

    | test <output_format> (=|~=) <output_format>
    | test-not <output_format> (=|~=) <output_format>

    | true

    | trigger-binding <binding> --mode <mode_id>
    | trigger-binding --mode <mode_id> <binding>

    | view [--toggle|--monitor <monitor_id>]... <tag>... [--toggle|--monitor <monitor_id>]...
    | view [--toggle|--monitor <monitor_id>]... --index <number>... [--toggle|--monitor <monitor_id>]...
    | view [--monitor <monitor_id>] --all [--monitor <monitor_id>]
    | view [--monitor <monitor_id>] [--back-and-forth] [--monitor <monitor_id>]

    | volume [--no-gui] (up|down) [--no-gui]
    | volume [--no-gui] (mute-toggle|mute-on|mute-off) [--no-gui]
    | volume [--no-gui] set <number> [--no-gui]

    | zoom [--window-id <window_id>|--monitor <monitor_id>]...

    | list-apps [--macos-native-hidden [no] | --format <output_format> | --count | --json]...

    | list-exec-env-vars

    | list-modes [--current|--json|--count]...

    | list-monitors [--focused [no] | --mouse [no] | --format <output_format> | --count | --json]...

    | list-tags [<list_tags_flag>]...

    | list-windows [<list_windows_filter_flag> | --format <output_format> | --count | --json]...
    | list-windows [--format <output_format>|--count|--json]... (--all|--focused) [--format <output_format>|--count|--json]...

    | subscribe [--all|--no-send-initial] <event>... [--all|--no-send-initial]
    ;

<window_id> ::= {{{ tomwm list-windows --all --format '%{window-id}%{tab}%{app-name} - %{window-title}' }}};
<binding> ::= {{{ tomwm config --get mode --keys | xargs -I{} tomwm config --get mode.{}.binding --keys }}};
<mode_id> ::= {{{ tomwm config --get mode --keys }}};
<tag> ::= {{{ tomwm list-tags --monitor all }}};
<number> ::= {{{ true }}};
<monitor_pattern> ::= {{{ true }}};
<layouts> ::= tile|monocle|floating;

<list_tags_flag> ::= --monitor <monitor_id>... | --selected [no] | --occupied [no]
    | --format <output_format> | --count | --json;

<move_to_monitor1_flag> ::= --window-id <window_id>|--focus-follows-window|--fail-if-noop|--wrap-around;
<move_to_monitor2_flag> ::= --window-id <window_id>|--focus-follows-window|--fail-if-noop;

<focus_direction_flag> ::= --boundaries <boundary>|--ignore-floating|--fail-if-noop;
<boundary> ::= monitor|all-monitors;

<list_windows_filter_flag> ::= --tag (selected | <tag>)...
    | --monitor <monitor_id>...
    | --pid <pid>
    | --app-bundle-id <app_bundle_id>
    ;
<output_format> ::= {{{ true }}};

<app_bundle_id> ::= {{{ tomwm list-apps --format '%{app-bundle-id}%{tab}%{app-name}' }}};
<pid> ::= {{{ tomwm list-apps --format '%{app-pid}%{tab}%{app-name}' }}};
<monitor_id> ::= all | mouse | focused | {{{ tomwm list-monitors --format '%{monitor-id}%{tab}%{monitor-name}' }}};

<event> ::= focus-changed | focused-monitor-changed | tags-changed | mode-changed | window-detected | binding-triggered;
